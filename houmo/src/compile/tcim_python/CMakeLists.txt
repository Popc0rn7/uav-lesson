cmake_minimum_required(VERSION 3.16)
project(aerovla_tcim_python LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  message(FATAL_ERROR "OpenHarmony toolchain required")
endif()
foreach(required TCIM_ROOT TARGET_PYTHON_INCLUDE PYBIND11_INCLUDE TARGET_PYTHON_LIBRARY)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
add_library(pytcim MODULE "${TCIM_ROOT}/python/pytcim.cc")
target_compile_features(pytcim PRIVATE cxx_std_17)
target_include_directories(pytcim PRIVATE "${TCIM_ROOT}/include" "${TARGET_PYTHON_INCLUDE}" "${PYBIND11_INCLUDE}")
target_link_directories(pytcim PRIVATE "${TCIM_ROOT}/lib")
target_link_libraries(pytcim PRIVATE tcim_runtime_lite "${TARGET_PYTHON_LIBRARY}")
set_target_properties(pytcim PROPERTIES PREFIX "" SUFFIX ".cpython-312-aarch64-linux-ohos.so"
  BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
