cmake_minimum_required(VERSION 3.16)
project(grounding_dino_ohos_infer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  message(FATAL_ERROR "OpenHarmony toolchain required")
endif()
add_executable(grounding_dino_infer main.cc)
target_include_directories(grounding_dino_infer PRIVATE "$ENV{TCIM_RUNTIME_PATH}/include")
target_link_directories(grounding_dino_infer PRIVATE "$ENV{TCIM_RUNTIME_PATH}/lib")
target_link_libraries(grounding_dino_infer PRIVATE tcim_runtime_lite)
set_target_properties(grounding_dino_infer PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
