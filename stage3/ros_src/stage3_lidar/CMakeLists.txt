cmake_minimum_required(VERSION 3.10)
project(stage3_lidar)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(prefilter src/prefilter.cpp)
target_compile_features(prefilter PRIVATE cxx_std_17)
ament_target_dependencies(prefilter rclcpp sensor_msgs)
target_link_libraries(prefilter gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS prefilter DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_executable(test_mid360_pattern test/test_pattern.cpp)
  target_compile_features(test_mid360_pattern PRIVATE cxx_std_17)
  target_include_directories(test_mid360_pattern PRIVATE src)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_mid360_pattern PRIVATE -UNDEBUG)
  endif()
  add_test(NAME mid360_pattern COMMAND test_mid360_pattern)
endif()
find_package(livox_ros_driver2 QUIET)
if(livox_ros_driver2_FOUND)
  add_executable(timed_lidar src/timed_lidar.cpp)
  target_compile_features(timed_lidar PRIVATE cxx_std_17)
  ament_target_dependencies(timed_lidar rclcpp sensor_msgs livox_ros_driver2)
  target_link_libraries(timed_lidar gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
  install(TARGETS timed_lidar DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
